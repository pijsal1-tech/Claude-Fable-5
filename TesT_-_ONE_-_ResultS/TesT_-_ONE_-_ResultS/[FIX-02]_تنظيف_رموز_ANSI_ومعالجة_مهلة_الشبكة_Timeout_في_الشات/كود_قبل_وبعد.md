```python
import re

def strip_ansi_codes(text: str) -> str:
    """TSK-502 / FIX-02: تنظيف أكواد ألوان ANSI الخام من أي نص خطأ قبل إرساله للواجهة"""
    if not text:
        return ""
    ansi_escape = re.compile(r'(?:\x1B[@-Z\\-_]|[\x80-\x9A\x9C-\x9F]|(?:\x1B\[|\x9B)[0-?]*[ -/]*[@-~])')
    cleaned = ansi_escape.sub('', text)
    cleaned = re.sub(r'\[\d{1,2}m', '', cleaned)
    return cleaned.strip()

except Exception as e:
    clean_err = strip_ansi_codes(str(e))
    if "timed out" in clean_err.lower() or "curl: (28)" in clean_err:
        friendly_msg = f"⚠️ تعذّر الاتصال بالمزود [{provider_name}]: انتهت مهلة الانتظار (Timeout) نظرًا لتأخر الاستجابة. يرجى المحاولة لاحقًا أو اختيار موديل آخر."
    else:
        friendly_msg = f"❌ [{provider_name}] خطأ في الاتصال: {clean_err}"
        
    sctx.send({
        "type": "error", 
        "text": friendly_msg
    })
```

```javascript
// دالة تنظيف ألوان الـ ANSI في الفرونت إند لدفاع إضافي
function cleanAnsiColors(text) {
    if (!text) return "";
    return text.replace(/[\u001b\u009b][[()#;?]*(?:[0-9]{1,4}(?:;[0-9]{0,4})*)?[0-9A-ORZcf-nqry=><]/g, '')
               .replace(/\[\d{1,2}m/g, '');
}

function handleErrorMessage(msg) {
    const errorDiv = document.createElement("div");
    errorDiv.className = "chat-error";
    errorDiv.textContent = cleanAnsiColors(msg.text); // نص عربي نظيف بدون رموش ألوان
    chatMessages.appendChild(errorDiv);
}
```
