# ═══ الكود بعد الإصلاح ═══
from pathlib import Path


def is_related_to_workspace(detected: Path, root: Path) -> bool:
    """
    هل المسار المكتشف له علاقة بشجرة المشروع المفتوح؟
    (نفس المسار / مجلد ولد / مجلد أب)

    ملحوظة مهمة: PureWindowsPath بيقارن case-insensitive أصلاً،
    فمش محتاجين normcase على Windows. والـ relative_to بيقارن
    مكونات المسار مش نصوص — يعني "e:\\app" مش هتطابق "e:\\apple".
    """
    d, r = detected.resolve(), root.resolve()

    if d == r:
        return True                          # نفس المشروع بالظبط

    for sub, base in ((d, r), (r, d)):
        try:
            sub.relative_to(base)            # واحد منهم جوه التاني؟
            return True                      # (ولد أو أب)
        except ValueError:
            continue

    return False                             # مسار غريب عن الشجرة


def _is_dangerous_root(path: Path) -> bool:
    """
    حماية إضافية: جذور الدرايفات ومجلدات النظام ممنوع يظهرلها كارت أصلًا —
    لأن مجرد وجود الزرار معناه إن حد ممكن يدوسه غلط (تشديد-فقط).
    """
    p = path.resolve()
    if p.parent == p:                        # جذر درايف: D:\ أو C:\ أو /
        return True
    dangerous = {"c:\\windows", "c:\\users", "c:\\program files"}
    return str(p).lower() in dangerous

# 🐛 تقرير مشكلة: كشف مسار خاطئ عند إرفاق ملف من نفس المشروع

## 📌 وصف المشكلة بالبلدي

في نظام الشات بتاعنا، لما المستخدم بيسحب ملف (Drag & Drop) أو يرفقه من **جوه نفس المشروع المفتوح حالياً**، النظام غلط بيكتشف مسار غريب (زي `D:\` لوحده) وبيطلع للمستخدم كارت سؤال يقوله "اكتشفت مسار مختلف، عايز تغير المجلد ولا تتجاهل؟" — رغم إن الملف ده أصلاً **جوه المشروع اللي فاتحه من الأساس**، ومفيش أي داعي إطلاقاً للسؤال ده.

المفروض النظام يفرق بين:
- مسار **جوه** المشروع الحالي → يتجاهل الفحص خالص، يكمل عادي
- مسار **برا** المشروع تماماً (مشروع تاني، درايف تاني) → هنا بس يظهر الكارت

لكن دلوقتي هو بيعامل الحالتين بنفس الطريقة وبيسأل في الحالتين، وده تجربة مستخدم سيئة جداً وبيبوظ الفلو الطبيعي.

## 🔍 مكان المشكلة المتوقع في الكود

- **الملف الرئيسي المشتبه فيه**: `core/chat_dispatch.py` — فيه المنطق اللي بيكتشف المسار من نص الرسالة (`detected_dir`) قبل ما يبعت `path_detected_options` (الكارت اللي فيه "تغيير مجلد العمل / إرفاق كسياق فقط / تجاهل والمتابعة").
- **ملف تاني موجود بالفعل وممكن يكون فيه نصف الحل جاهز**: `chain/path_policy.py` — فيه دالة `resolve_workspace_path` مستخدمة في مكان تاني من المشروع (حماية من Zip-Slip في استعادة الباك-أب). الدالة دي أصلاً بتعمل "فحص احتواء" (containment check) — يعني تتأكد إن مسار معين واقع جوه جذر تاني ولا لأ. **محتمل تقدر تعيد استخدامها هنا بدل ما تكتب دالة فحص جديدة من الصفر.**

## ⚠️ حاجات لازم تتفحص فعلياً في الكود قبل ما نطبق أي حل (مهم جداً)

قبل ما تطبق أي fix، افحص الحاجات دي بالظبط لأن الحل بيختلف حسب الإجابة:

### 1. إزاي بيتم استخراج `detected_dir` من نص الرسالة؟
دور على الدالة أو الـ regex اللي بيستخرج المسار من نص المستخدم (غالباً جوه `core/chat_dispatch.py` أو موديول مساعد بيستدعيه). **السؤال المهم**: هل الدالة دي بتقتص المسار وتاخد أول جزء منه بس (زي `D:\`) بدل المسار الكامل (`D:\SMS\...\file.py`)؟ لو الإجابة أيوه، **المشكلة الحقيقية هنا مش في المقارنة** — المشكلة في الاستخراج نفسه، ولازم يتصلح الأول قبل أي حاجة تانية.

### 2. هل فيه تعارض بين نظام الإرفاق الرسمي (attachment) ونظام كشف المسار في النص؟
لو الرسالة أصلاً بتحمل حقل `attachments` أو مشابه (زي اللي شفناه في الواجهة "تم إرفاق: filename.py")، **المفروض من الأساس فحص المسار في النص ميتنفذش خالص** لو فيه attachment رسمي متسجل. دور على المكان اللي بيتحدد فيه `skip_path_detection` وشوف هل بيتفعّل تلقائي مع وجود attachment ولا لأ.

### 3. هل `chain/path_policy.py` فيه دالة جاهزة تصلح لإعادة الاستخدام؟
افحص الموديول ده كامل وشوف الـ signature بتاع `resolve_workspace_path` أو أي دالة مشابهة — لو منطقها قريب من اللي محتاجينه، استخدمها بدل تكرار المنطق في مكانين مختلفين من المشروع.

# core/chat_dispatch.py
import os
from pathlib import Path

def is_related_to_workspace(detected: Path, root: Path) -> bool:
    """
    يتأكد إن المسار المكتشف جزء من شجرة المشروع الحالي —
    سواء نفس المسار أو مجلد فرعي (ابن) منه.
    ملحوظة: حالة "الأب" (زي D:\ كامل) متتعاملش تلقائياً كـ"آمن" —
    لازم تتفحص الأول (راجع نقطة الاستخراج فوق) عشان معرفش
    نخفي مشاكل حقيقية بالغلط.
    """
    try:
        d = os.path.normcase(str(detected.resolve()))
        r = os.path.normcase(str(root.resolve()))
    except (OSError, ValueError):
        return False  # مسار مشوّه = افترض غير مرتبط (أأمن سلوك)

    if d == r:
        return True  # نفس المشروع بالظبط

    try:
        Path(d).relative_to(Path(r))
        return True  # المسار المكتشف جوه (ابن) المشروع
    except ValueError:
        return False  # مسار غريب فعلاً — هنا بس نسأل


# في مكان الفحص:
has_official_attachment = bool(msg.get("attachments"))

if detected_dir and not skip_path_detection and not has_official_attachment:
    if not is_related_to_workspace(Path(detected_dir), Path(sctx.fm.root)):
        store_pending_path_request(req_id, {
            "path": detected_dir,
            "user_text": user_text,
            "mode": mode,
            "msg": msg,
        })
        sctx.send({
            "type": "path_detected_options",
            "path": detected_dir,
            "request_id": req_id,
        })
        return
    # وإلا: المسار جوه المشروع — نكمل عادي بدون أي كارت

# 📝 الملخص المباشر للمشكلة (بالبلدي كده)

### 📌 إيه المشكلة بالضبط؟
المحرر فيه ميزة: لما تسحب ملف في الشات أو تكتب مسار، المحرر بيكتشف المسار ويرفع لك "كارت سؤال" يقولك: *"هل حابب تفتح وتنتقل للمشروع ده؟"*.

### 🧩 فين الـ Bug الخفي والخلل المنطقي؟
1. **الـ Regex مقطوع**: الـ Regex بتاع كشف مسارات الويندوز فيه مسافة `[\\/ ]`! فلما تسحب ملف زي `D:\SMS\project\file.txt` المحرر بياخد `D:\` بس! وبيفكر إنك كاتب الهارد كله `D:\`!
2. **عمياء عن موقعك الحالي**: المحرر أول ما بيشوف أي مسار، بيرفع كارت السؤال فوراً، **من غير ما يسأل نفسه**: *"هو الملف ده مش جوه مجلد مشروعي المفتوح حالياً أصلاً؟"* فلو الملف جوه مشروعك أو مجلد فرعي منه، عيب ومظهره مش حلو يسألك تغير المشروع وأنت واقف جواه!
3. **مبيفرقش بين السحب الصريح والكتابة**: لو الملف مسحوب كمرفق صريح (`attachments`)، المفروض المحرر يقرأه على طول مش يروح يدور على مسارات نصية في الشات!

import os
from pathlib import Path

# ✅ 1. دالة حظر جذور المحركات ومجلدات النظام (منع اقتراح فتح D:\ أو C:\Windows)
def is_root_or_system_dir(p: Path) -> bool:
    resolved = p.resolve()
    # إذا كان المسار هو root المحرك نفسه (مثل D:\ أو C:\)
    if resolved.parent == resolved or len(resolved.parts) <= 1:
        return True
    norm = os.path.normcase(str(resolved))
    if norm in [os.path.normcase(r"C:\windows"), os.path.normcase(r"C:\users")]:
        return True
    return False

# ✅ 2. دالة الفحص الذكية للتأكد هل المسار مرتبط بالمشروع المفتوح
def is_related_to_workspace(detected_path: str, project_root: str) -> bool:
    try:
        detected = Path(detected_path).resolve()
        root = Path(project_root).resolve()
        
        # حظر درايف الجذر أو مجلدات النظام -> تجاهل صامت
        if is_root_or_system_dir(detected):
            return True
            
        # فحص المساواة الصريحة بـ Path (نفس المجلد)
        if detected == root:
            return True
            
        # فحص التبعية المتبادلة (سواء مجلد فرعي أو أب للمشروع) بـ relative_to النظيفة
        for sub, base in ((detected, root), (root, detected)):
            try:
                sub.relative_to(base)
                return True
            except ValueError:
                continue
                
        return False
    except Exception:
        return False

# ✅ 3. التعديل المباشر داخل منطق ChatDispatch:
# -------------------------------------------------------------

if (msg.get("attachments") or msg.get("from_attachment")) and not skip_path_detection:
    skip_path_detection = True

# ب. الـ Regex المضبوط (بدون مسافة قاطعة)
win_paths = re.findall(r'[A-Za-z]:[\\/][^\s,;"\'>]+', user_text)

# ج. شرط الفحص الذكي قبل رفع كارت السؤال:

    current_project_root = getattr(sctx.fm, "root", "")
    
    # لو المسار يتبع مشروعك الحالي أو ابن له أو جذر الدرايف -> اسكوت تماماً وكمّل قراءة الملف!
    if is_related_to_workspace(detected_dir, current_project_root):
        pass  # تجاهل صامت ومتابعة طبيعية
    else:
        # مسار خارجي غريب عن المشروع فعلاً -> ارفع كارت السؤال

        deps.store_pending_path_request(req_id, {...})
        sctx.send({
            "type": "path_detected_options",
            "request_id": req_id,
            "path": detected_dir
        })
